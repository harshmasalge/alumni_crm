// ── API HANDLER ───────────────────────────────────────────────
function handleRequest(action, dataJson) {
  try {
    var d = dataJson ? JSON.parse(dataJson) : {};
    if (action === 'getAll')          return ok(getAllData());
    if (action === 'saveAlumni')      return ok(saveAlumni(d));
    if (action === 'deleteAlumni')    return ok(deleteAlumni(d.roll));
    if (action === 'saveDonation')    return ok(saveDonationRow(d));
    if (action === 'deleteDonation')  return ok(deleteDonationRow(d.roll));
    if (action === 'saveEvent')       return ok(upsertExtra('Events', d, '_id'));
    if (action === 'deleteEvent')     return ok(deleteEventFull(d.id));
    if (action === 'saveAttendance')  return ok(saveAttendance(d.eventId, d.rolls));
    if (action === 'saveJobChange')   return ok(saveJobChange(d));
    if (action === 'markVerified')    return ok(markVerified(d.roll));
    return err('Unknown action: ' + action);
  } catch(e) {
    return err(e.message);
  }
}
function ok(p)  { return JSON.stringify({ ok:true,  payload:p }); }
function err(m) { return JSON.stringify({ ok:false, error:m   }); }

// ── READ ALUMNI ───────────────────────────────────────────────
function readAlumni() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(ALUMNI_SHEET);
  if (!sheet) throw new Error('Sheet "'+ALUMNI_SHEET+'" not found. Edit ALUMNI_SHEET in Code.gs.');
  var vals = sheet.getDataRange().getValues();
  if (vals.length <= 1) return [];
  var headers = vals[0].map(function(h){ return String(h).trim(); });
  var result  = [];
  for (var i = 1; i < vals.length; i++) {
    var row = vals[i];
    if (row.every(function(c){ return c===''||c===null; })) continue;
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = (row[j]===null||row[j]===undefined) ? '' : String(row[j]);
    }
    if (!obj['_id']||obj['_id']==='')
      obj['_id'] = 'alm_'+String(obj['Roll No']||i).replace(/[^a-zA-Z0-9]/g,'')+'-'+i;
    if (!obj['_color']||obj['_color']==='')
      obj['_color'] = COLORS[i % COLORS.length];
    result.push(obj);
  }
  return result;
}

// ── SAVE DONATION ROW ─────────────────────────────────────────
// Upsert by Roll No — updates existing row or appends new one
function saveDonationRow(data) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(DONATIONS_SHEET);
  if (!sheet) throw new Error('Donations sheet "'+DONATIONS_SHEET+'" not found.');
  var vals    = sheet.getDataRange().getValues();
  var headers = vals[0].map(function(h){ return String(h).trim(); });
  var rollIdx = headers.indexOf('Roll No');
  if (rollIdx < 0) throw new Error('"Roll No" column not found in Donations sheet.');
  var rollVal = String(data['Roll No']||'').trim();
  var rowData = headers.map(function(h){ var v=data[h]; return (v!==undefined&&v!==null)?v:''; });
  // Recalculate Total
  var totalIdx = headers.indexOf('Total');
  if (totalIdx >= 0) {
    var total = 0;
    DON_CATEGORIES.forEach(function(cat){
      var idx = headers.indexOf(cat);
      if (idx >= 0) total += Number(rowData[idx])||0;
    });
    rowData[totalIdx] = total;
  }
  // Update if exists
  for (var i = 1; i < vals.length; i++) {
    if (String(vals[i][rollIdx]).trim() === rollVal) {
      sheet.getRange(i+1, 1, 1, headers.length).setValues([rowData]);
      return 'updated';
    }
  }
  // Append new
  sheet.appendRow(rowData);
  return 'inserted';
}


// ── DELETE DONATION ROW ───────────────────────────────────────
function deleteDonationRow(roll) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(DONATIONS_SHEET);
  if (!sheet) return 'not_found';
  var vals    = sheet.getDataRange().getValues();
  var headers = vals[0].map(function(h){ return String(h).trim(); });
  var rollIdx = headers.indexOf('Roll No');
  for (var i = vals.length-1; i >= 1; i--) {
    if (String(vals[i][rollIdx]).trim() === String(roll).trim()) {
      // Clear donation amounts but keep the row (preserve alumni info)
      var rowData = vals[i].slice();
      DON_CATEGORIES.forEach(function(cat){
        var idx = headers.indexOf(cat);
        if (idx >= 0) rowData[idx] = '';
      });
      var donIdx = headers.indexOf('Donation in 24-25');
      if (donIdx >= 0) rowData[donIdx] = '';
      var totalIdx = headers.indexOf('Total');
      if (totalIdx >= 0) rowData[totalIdx] = '';
      sheet.getRange(i+1, 1, 1, headers.length).setValues([rowData]);
      return 'cleared';
    }
  }
  return 'not_found';
}


// ── SAVE ALUMNI ───────────────────────────────────────────────
function saveAlumni(data) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(ALUMNI_SHEET);
  if (!sheet) throw new Error('Sheet "'+ALUMNI_SHEET+'" not found.');
  var vals    = sheet.getDataRange().getValues();
  var headers = vals[0].map(function(h){ return String(h).trim(); });
  // Ensure internal columns exist
  for (var k=0; k<INTERNAL_COLS.length; k++) {
    if (headers.indexOf(INTERNAL_COLS[k]) === -1) {
      var nc = headers.length+1;
      sheet.getRange(1,nc).setValue(INTERNAL_COLS[k]).setFontWeight('bold').setBackground('#fce8d5');
      headers.push(INTERNAL_COLS[k]);
    }
  }
  var rollIdx = headers.indexOf('Roll No');
  var rollVal = String(data['Roll No']||'').trim();
  var rowData = headers.map(function(h){ var v=data[h]; return (v!==undefined&&v!==null)?v:''; });
  var idIdx    = headers.indexOf('_id');
  var colorIdx = headers.indexOf('_color');
  if (idIdx>=0 && (!rowData[idIdx]||rowData[idIdx]===''))
    rowData[idIdx] = 'alm_'+rollVal.replace(/[^a-zA-Z0-9]/g,'')+'-'+Date.now();
  if (colorIdx>=0 && (!rowData[colorIdx]||rowData[colorIdx]===''))
    rowData[colorIdx] = COLORS[vals.length % COLORS.length];
  for (var i=1; i<vals.length; i++) {
    if (String(vals[i][rollIdx]).trim() === rollVal) {
      sheet.getRange(i+1,1,1,headers.length).setValues([rowData]);
      return 'updated';
    }
  }
  sheet.appendRow(rowData);
  return 'inserted';
}


function deleteAlumni(roll) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(ALUMNI_SHEET);
  if (!sheet) return 'not_found';
  var vals    = sheet.getDataRange().getValues();
  var headers = vals[0].map(function(h){ return String(h).trim(); });
  var rollIdx = headers.indexOf('Roll No');
  for (var i=vals.length-1; i>=1; i--) {
    if (String(vals[i][rollIdx]).trim()===String(roll).trim()) {
      sheet.deleteRow(i+1); return 'deleted';
    }
  }
  return 'not_found';
}


function markVerified(roll) {
  var all = readAlumni();
  for (var i=0; i<all.length; i++) {
    if (String(all[i]['Roll No']).trim()===String(roll).trim()) {
      all[i]['_lastVerified'] = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
      saveAlumni(all[i]); return 'saved';
    }
  }
  return 'not_found';
}


function saveJobChange(data) {
  upsertExtra('JobHistory', data.history, '_id');
  var all = readAlumni();
  for (var i=0; i<all.length; i++) {
    if (String(all[i]['Roll No']).trim()===String(data.roll).trim()) {
      var a = all[i];
      if (data.designation)  a['Designation']                       = data.designation;
      if (data.organization) a['Organization/Institute/University'] = data.organization;
      if (data.city)         a['City']                              = data.city;
      if (data.state)        a['State/ Province']                   = data.state;
      if (data.country)      a['Country']                           = data.country;
      if (data.sector)       a['Sector']                            = data.sector;
      if (data.joiningDate)  a['Month and Year of Joining']         = data.joiningDate;
      a['_lastVerified'] = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
      saveAlumni(a); return 'saved';
    }
  }
  return 'not_found';
}


// ── EXTRA SHEETS ──────────────────────────────────────────────
function getOrCreateExtra(name) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    var hdrs = EXTRA[name];
    sheet.appendRow(hdrs);
    sheet.getRange(1,1,1,hdrs.length).setFontWeight('bold').setBackground('#c9daf8');
    sheet.setFrozenRows(1);
  }
  return sheet;
}


function readExtra(name) {
  var sheet = getOrCreateExtra(name);
  var vals  = sheet.getDataRange().getValues();
  if (vals.length<=1) return [];
  var keys = vals[0]; var result = [];
  for (var i=1; i<vals.length; i++) {
    if (vals[i].every(function(c){ return c===''; })) continue;
    var obj = {};
    for (var j=0; j<keys.length; j++) obj[keys[j]] = (vals[i][j]===''||vals[i][j]===null)?null:vals[i][j];
    result.push(obj);
  }
  return result;
}


function upsertExtra(name, obj, keyCol) {
  var hdrs   = EXTRA[name];
  var sheet  = getOrCreateExtra(name);
  var vals   = sheet.getDataRange().getValues();
  var keyIdx = vals[0].indexOf(keyCol);
  var keyVal = obj[keyCol]||obj['_id']||obj['id'];
  var rowData = hdrs.map(function(h){ var v=obj[h]; return (v!==undefined&&v!==null)?v:''; });
  for (var i=1; i<vals.length; i++) {
    if (String(vals[i][keyIdx])===String(keyVal)) {
      sheet.getRange(i+1,1,1,hdrs.length).setValues([rowData]); return 'updated';
    }
  }
  sheet.appendRow(rowData); return 'inserted';
}


function deleteEventFull(id) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ev = ss.getSheetByName('Events');
  if (ev) {
    var vals = ev.getDataRange().getValues();
    var idx  = vals[0].indexOf('_id');
    for (var i=vals.length-1; i>=1; i--) {
      if (String(vals[i][idx])===String(id)) ev.deleteRow(i+1);
    }
  }
  var att = ss.getSheetByName('Attendance');
  if (att) {
    var av2 = att.getDataRange().getValues();
    for (var j=av2.length-1; j>=1; j--) {
      if (String(av2[j][0])===String(id)) att.deleteRow(j+1);
    }
  }
  return 'deleted';
}


function saveAttendance(eventId, rolls) {
  var sheet = getOrCreateExtra('Attendance');
  var vals  = sheet.getDataRange().getValues();
  for (var i=vals.length-1; i>=1; i--) {
    if (String(vals[i][0])===String(eventId)) sheet.deleteRow(i+1);
  }
  rolls.forEach(function(r){ sheet.appendRow([eventId, r]); });
  return 'saved';
}

// ── CHECK BIRTHDAYS & SEND EMAIL (runs daily automatically) ──
function checkBirthdays() {
  var today    = new Date();
  var todayDay = today.getDate();
  var todayMon = today.getMonth() + 1; // 1-based


  var alumni = readAlumni();
  var bdays  = [];


  alumni.forEach(function(a) {
    var dob = a['DOB'];
    if (!dob) return;
    try {
      var d = new Date(dob);
      if (isNaN(d.getTime())) return;
      if (d.getDate() === todayDay && (d.getMonth()+1) === todayMon) {
        bdays.push(a);
      }
    } catch(e) {}
  });


  if (bdays.length === 0) return; // no birthdays today, no email sent


  sendBirthdayEmail(bdays, today);
}

// ── ADD DOB COLUMN TO YOUR SHEET ─────────────────────────────
function addDobColumn() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(ALUMNI_SHEET);
  if (!sheet) { SpreadsheetApp.getUi().alert('Sheet not found!'); return; }


  var headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0].map(function(h){return String(h).trim();});
  if (headers.indexOf('DOB') >= 0) {
    SpreadsheetApp.getUi().alert('✅ DOB column already exists in column ' + (headers.indexOf('DOB')+1));
    return;
  }
  var newCol = sheet.getLastColumn() + 1;
  sheet.getRange(1, newCol).setValue('DOB').setFontWeight('bold').setBackground('#fce8d5');


  // Set date format for the whole column
  sheet.getRange(2, newCol, sheet.getMaxRows()-1, 1).setNumberFormat('dd/mm/yyyy');


  SpreadsheetApp.getUi().alert(
    '✅ DOB column added to column ' + newCol + '!\n\n'
    + 'Enter dates in format: DD/MM/YYYY\n'
    + 'Example: 15/08/1998\n\n'
    + 'Next: Run setupBirthdayTrigger() to activate daily email reminders.'
  );
}


// ── SET UP DAILY TRIGGER (run this once) ─────────────────────
function setupBirthdayTrigger() {

// ================================================================
//  BULK IMPORT FROM EXTERNAL GOOGLE SPREADSHEET
//
//  HOW TO USE:
//  1. Paste the URL of your external spreadsheet into IMPORT_SHEET_URL below
//  2. Set IMPORT_TAB_NAME to the tab name inside that spreadsheet
//  3. Run importFromExternalSheet() from the function dropdown
//  4. A summary will tell you what was updated / added
//
//  RULES:
//  - Matching is done by Roll No
//  - Only non-empty cells in the source overwrite existing values
//    (so if source has blank City, existing City is kept)
//  - If Roll No not found → new alumni row is added
//  - Job/Location changes are auto-logged to JobHistory sheet
// ================================================================


var IMPORT_SHEET_URL = 'PASTE_YOUR_GOOGLE_SHEET_URL_HERE';
var IMPORT_TAB_NAME  = 'Sheet1'; // tab name inside the external spreadsheet


// Column mapping: what your SOURCE sheet columns are called
// Change these if your external sheet uses different header names
var IMPORT_COL_MAP = {
  'Roll No':                           'Roll No',
  'YOG':                               'YOG',
  'Course':                            'Course',
  'Name of the alumni':                'Name of the alumni',
  'Designation':                       'Designation',
  'Organization/Institute/University': 'Organization/Institute/University',
  'Sector':                            'Sector',
  'City':                              'City',
  'State/ Province':                   'State/ Province',
  'Country':                           'Country',
  'Month and Year of Joining':         'Month and Year of Joining',
  'Linkedin':                          'Linkedin',
  'IITGn Email':                       'IITGn Email',
  'Personal Email ID':                 'Personal Email ID',
  'Work Email ID':                     'Work Email ID'
};


function importFromExternalSheet() {
  // ── Validate URL ──
  if (!IMPORT_SHEET_URL || IMPORT_SHEET_URL === 'PASTE_YOUR_GOOGLE_SHEET_URL_HERE') {
    SpreadsheetApp.getUi().alert(
      '❌ Please paste your Google Sheet URL into the\nIMPORT_SHEET_URL variable in Code.gs first.'
    );
    return;
  }


  // ── Open external spreadsheet ──
  var extSS, extSheet;
  try {
    extSS    = SpreadsheetApp.openByUrl(IMPORT_SHEET_URL);
    extSheet = extSS.getSheetByName(IMPORT_TAB_NAME);
    if (!extSheet) {
      SpreadsheetApp.getUi().alert(
        '❌ Tab "' + IMPORT_TAB_NAME + '" not found in the external spreadsheet.\n'
        + 'Check IMPORT_TAB_NAME in Code.gs.\n\n'
        + 'Available tabs: ' + extSS.getSheets().map(function(s){return s.getName();}).join(', ')
      );
      return;
    }
  } catch(e) {
    SpreadsheetApp.getUi().alert(
      '❌ Could not open the spreadsheet.\n\n'
      + 'Make sure:\n'
      + '1. The URL is correct\n'
      + '2. You have access to that spreadsheet\n'
      + '3. The spreadsheet is shared with your Google account\n\n'
      + 'Error: ' + e.message
    );
    return;
  }


  // ── Read external sheet ──
  var extVals = extSheet.getDataRange().getValues();
  if (extVals.length <= 1) {
    SpreadsheetApp.getUi().alert('⚠ The external sheet appears to be empty (only headers or no data).');
    return;
  }


  var extHeaders = extVals[0].map(function(h){ return String(h).trim(); });
  var extRows    = extVals.slice(1);


  // ── Read current alumni ──
  var currentAlumni = readAlumni();
  var currentByRoll = {};
  currentAlumni.forEach(function(a){
    currentByRoll[String(a['Roll No']).trim()] = a;
  });


  var updated = 0, added = 0, skipped = 0;
  var changeLog = [];
  var today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');


  extRows.forEach(function(row) {
    // skip fully empty rows
    if (row.every(function(c){ return c === '' || c === null; })) return;


    // Build object from external row using column mapping
    var incoming = {};
    Object.keys(IMPORT_COL_MAP).forEach(function(crmCol) {
      var srcCol = IMPORT_COL_MAP[crmCol];
      var idx    = extHeaders.indexOf(srcCol);
      if (idx >= 0 && row[idx] !== '' && row[idx] !== null && row[idx] !== undefined) {
        incoming[crmCol] = String(row[idx]).trim();
      }
    });


    var roll = incoming['Roll No'];
    if (!roll) { skipped++; return; }


    var existing = currentByRoll[roll];


    if (existing) {
      // ── UPDATE existing alumni ──
      var changed = false;
      var jobChanged = false;


      // Track job/location change fields
      var jobFields = ['Designation','Organization/Institute/University','City','State/ Province','Country','Sector','Month and Year of Joining'];
      var oldJob = {};
      jobFields.forEach(function(f){ oldJob[f] = existing[f] || ''; });


      // Apply incoming values (only overwrite if incoming is non-empty)
      Object.keys(incoming).forEach(function(col) {
        if (col === 'Roll No') return; // never overwrite the key
        if (incoming[col] && incoming[col] !== (existing[col]||'')) {
          existing[col] = incoming[col];
          changed = true;
          if (jobFields.indexOf(col) >= 0) jobChanged = true;
        }
      });


      if (changed) {
        existing['_lastVerified'] = today;
        saveAlumni(existing);
        updated++;


        // Auto-log job change if job/location fields changed
        if (jobChanged) {
          var histRecord = {
            _id:          'imp_' + roll.replace(/[^a-zA-Z0-9]/g,'') + '_' + Date.now(),
            alumniRoll:   roll,
            changeDate:   today,
            designation:  existing['Designation']  || '',
            organization: existing['Organization/Institute/University'] || '',
            city:         existing['City']         || '',
            state:        existing['State/ Province'] || '',
            country:      existing['Country']      || '',
            sector:       existing['Sector']       || '',
            joiningDate:  existing['Month and Year of Joining'] || '',
            notes:        'Auto-imported from external sheet'
          };
          upsertExtra('JobHistory', histRecord, '_id');
          changeLog.push(existing['Name of the alumni'] || roll);
        }
      }


    } else {
      // ── ADD new alumni ──
      var newAlumni = {
        _id:           'alm_' + roll.replace(/[^a-zA-Z0-9]/g,'') + '_' + Date.now(),
        _color:        COLORS[currentAlumni.length % COLORS.length],
        _lastVerified: today,
        _notes:        'Added via bulk import'
      };
      // Fill all CRM columns
      Object.keys(IMPORT_COL_MAP).forEach(function(crmCol) {
        newAlumni[crmCol] = incoming[crmCol] || '';
      });
      saveAlumni(newAlumni);
      currentAlumni.push(newAlumni);
      currentByRoll[roll] = newAlumni;
      added++;
    }
  });


  // ── Summary ──
  var summary = '✅ Import Complete!\n\n'
    + '📊 Results:\n'
    + '• ' + updated + ' alumni updated\n'
    + '• ' + added   + ' new alumni added\n'
    + '• ' + skipped + ' rows skipped (no Roll No)\n\n';


  if (changeLog.length > 0) {
    summary += '🔄 Job/location changes auto-logged for:\n'
      + changeLog.slice(0,10).map(function(n){ return '  · '+n; }).join('\n')
      + (changeLog.length > 10 ? '\n  · ...and '+(changeLog.length-10)+' more' : '') + '\n\n';
  }


  summary += 'Refresh the CRM in your browser to see the updates.';
  SpreadsheetApp.getUi().alert(summary);
}


// ── PREVIEW: see what will change before actually importing ──
function previewImport() {
  if (!IMPORT_SHEET_URL || IMPORT_SHEET_URL === 'PASTE_YOUR_GOOGLE_SHEET_URL_HERE') {
    SpreadsheetApp.getUi().alert('❌ Please set IMPORT_SHEET_URL in Code.gs first.');
    return;
  }


  var extSS, extSheet;
  try {
    extSS    = SpreadsheetApp.openByUrl(IMPORT_SHEET_URL);
    extSheet = extSS.getSheetByName(IMPORT_TAB_NAME);
    if (!extSheet) { SpreadsheetApp.getUi().alert('❌ Tab "'+IMPORT_TAB_NAME+'" not found.'); return; }
  } catch(e) {
    SpreadsheetApp.getUi().alert('❌ Could not open spreadsheet: ' + e.message);
    return;
  }


  var extVals   = extSheet.getDataRange().getValues();
  if (extVals.length <= 1) { SpreadsheetApp.getUi().alert('⚠ External sheet is empty.'); return; }


  var extHeaders = extVals[0].map(function(h){ return String(h).trim(); });
  var extRows    = extVals.slice(1).filter(function(r){ return !r.every(function(c){ return c===''; }); });


  var currentAlumni = readAlumni();
  var currentRolls  = {};
  currentAlumni.forEach(function(a){ currentRolls[String(a['Roll No']).trim()] = true; });


  var toUpdate = 0, toAdd = 0, toSkip = 0;
  var rollIdx  = extHeaders.indexOf('Roll No');


  extRows.forEach(function(row) {
    if (rollIdx < 0 || !row[rollIdx]) { toSkip++; return; }
    var roll = String(row[rollIdx]).trim();
    if (!roll) { toSkip++; return; }
    if (currentRolls[roll]) toUpdate++; else toAdd++;
  });


  SpreadsheetApp.getUi().alert(
    '👀 Import Preview\n\n'
    + 'External sheet: ' + extSheet.getName() + '\n'
    + 'Total rows found: ' + extRows.length + '\n\n'
    + '📊 What will happen:\n'
    + '• ' + toUpdate + ' existing alumni will be updated\n'
    + '• ' + toAdd    + ' new alumni will be added\n'
    + '• ' + toSkip   + ' rows will be skipped (no Roll No)\n\n'
    + 'Run importFromExternalSheet() to proceed.'
  );
}

function applyManualFilters(){
  return DB.alumni.filter(function(a){
    if(AF.name   &&(a['Name of the alumni']||'').toLowerCase().indexOf(AF.name.toLowerCase())<0)return false;
    if(AF.roll   &&(a['Roll No']||'').toLowerCase().indexOf(AF.roll.toLowerCase())<0)return false;
    if(AF.yog    &&String(a['YOG'])!==String(AF.yog))return false;
    if(AF.course &&a['Course']!==AF.course)return false;
    if(AF.sector &&a['Sector']!==AF.sector)return false;
    if(AF.org    &&(a['Organization/Institute/University']||'').toLowerCase().indexOf(AF.org.toLowerCase())<0)return false;
    if(AF.city   &&(a['City']||'').toLowerCase().indexOf(AF.city.toLowerCase())<0)return false;
    if(AF.country&&(a['Country']||'').toLowerCase().indexOf(AF.country.toLowerCase())<0)return false;
    if(AF.kw){var b=Object.values(a).join(' ').toLowerCase();if(b.indexOf(AF.kw.toLowerCase())<0)return false;}
    // Profile % filter
    if(AF.minPct!==''||AF.maxPct!==''){
      var pct=profPct(a);
      if(AF.minPct!==''&&pct<Number(AF.minPct))return false;
      if(AF.maxPct!==''&&pct>Number(AF.maxPct))return false;
    }
    // Engagement score filter
    if(AF.minEng!==''||AF.maxEng!==''){
      var es=eng(a['Roll No']);
      if(AF.minEng!==''&&es<Number(AF.minEng))return false;
      if(AF.maxEng!==''&&es>Number(AF.maxEng))return false;
    }
    return true;
  });
}


// ── Debounce: waits 300ms after typing stops before filtering ─
// This completely fixes the typing bug — page never re-renders mid-type
var _dbt=null;
function dbF(key,val){
  AF[key]=val;
  clearTimeout(_dbt);
  _dbt=setTimeout(function(){ updateSearchTable(); },300);
}


// ── Dropdowns update instantly (no debounce needed) ──────────
function fNow(key,val){
  AF[key]=val;
  updateSearchTable();
}


// ── Update ONLY the results table, never the inputs ──────────
// This is the key fix — inputs stay focused, only results change
function updateSearchTable(){
  var data=applyManualFilters();
  var countEl=gi('mf-count');
  if(countEl)countEl.textContent=data.length+' alumni found';
  var tbl=gi('search-results');
  if(!tbl)return;
  if(data.length){
    tbl.innerHTML='<div class="tw"><table><thead><tr>'
      +'<th>Name</th><th>Roll No</th><th>YOG</th><th>Course</th><th>Designation</th><th>Organization</th><th>Location</th><th>Sector</th><th></th>'
      +'</tr></thead><tbody>'
      +data.map(function(a){

          +'<td><div class="nr">'+av(a)+'<div><div style="font-weight:500">'+a['Name of the alumni']+'</div>'

  }
}

function clearAIResult(){
  AI_RESULT=null;AI_QUERY='';

  render();
}
async function runAISearch(){
  var el=gi('ai-q');
  var q=el?el.value.trim():AI_QUERY;
  if(!q)return;
  AI_QUERY=q;AI_LOADING=true;AI_RESULT=null;
  render();
  setTimeout(function(){
    var result=smartLocalSearch(q);
    AI_RESULT=result;AI_LOADING=false;
    render();
    setTimeout(function(){var inp=gi('ai-q');if(inp){inp.value=AI_QUERY;inp.focus();}},50);
  },200);
}

      if(tok.length<3||stopWords.has(tok)) return;

      if(org.indexOf(tok)>=0){score+=35;reasons.push('org:'+tok);}

    if(HARD_EXCLUDE) score=-9999;


    return{a:a, score:score, reasons:Array.from(new Set(reasons)), dt:dt, ec:ec, sw:sw, yrs:yrs, engS:engS};
  });


  // ════════════════════════════════════════════════════════════

  var results = scored.filter(function(x){return x.score>=20;});


  // Sort
  if(sortByDon||wantHighDonors)       results.sort(function(a,b){return b.dt-a.dt;});
  else if(sortByEng||wantMostEngaged) results.sort(function(a,b){return b.engS-a.engS;});
  else if(wantLeastEngaged)           results.sort(function(a,b){return a.engS-b.engS;});
  else if(sortByEvents)               results.sort(function(a,b){return b.ec-a.ec;});
  else if(sortByBatch)                results.sort(function(a,b){return a.a['YOG']-b.a['YOG'];});
  else if(sortByBatchDesc)            results.sort(function(a,b){return b.a['YOG']-a.a['YOG'];});
  else                                results.sort(function(a,b){return b.score-a.score;});


  if(topAsc) results.reverse();
  if(topN)   results = results.slice(0,topN);


  var rolls = results.map(function(x){return x.a['Roll No'];});


  // ════════════════════════════════════════════════════════════

  // ════════════════════════════════════════════════════════════
  var summary='';

  } else {
    var parts=[];

function saveAlumni(){
  var name=gi('a-name').value.trim(),roll=gi('a-roll').value.trim(),yog=gi('a-yog').value.trim();
  if(!name||!roll||!yog){toast('Name, Roll No and YOG are required','bad');return;}
  var existing=EDIT_ROLL?DB.alumni.filter(function(x){return x['Roll No']==EDIT_ROLL;})[0]:null;
  var rec=existing?Object.assign({},existing):{_id:'',_color:COLORS[DB.alumni.length%COLORS.length],_lastVerified:'',_notes:''};
  rec['YOG']=yog;rec['Roll No']=roll;rec['Course']=gi('a-course').value.trim();
  rec['Name of the alumni']=name;rec['Designation']=gi('a-desig').value.trim();
  rec['Organization/Institute/University']=gi('a-org').value.trim();rec['Sector']=gi('a-sector').value;
  rec['City']=gi('a-city').value.trim();rec['State/ Province']=gi('a-state').value.trim();
  rec['Country']=gi('a-country').value.trim();rec['Month and Year of Joining']=gi('a-joining').value.trim();
  rec['Linkedin']=gi('a-linkedin').value.trim();rec['IITGn Email']=gi('a-iitgn').value.trim();
  rec['Personal Email ID']=gi('a-personal').value.trim();rec['Work Email ID']=gi('a-work').value.trim();
  rec['_notes']=gi('a-notes').value.trim();
  rec['_phone']=gi('a-phone').value.trim();rec['_friend']=gi('a-friend').value.trim();
  rec['_hometown']=gi('a-hometown').value.trim();rec['_awards']=gi('a-awards').value.trim();
  rec['_photo']=gi('a-photo').value.trim();rec['_dob']=gi('a-dob').value.trim();
  rec['_gender']=gi('a-gender').value;rec['_marital']=gi('a-marital').value;
  if(EDIT_ROLL){var idx=DB.alumni.findIndex(function(x){return x['Roll No']==EDIT_ROLL;});if(idx>=0)DB.alumni[idx]=rec;}
  else DB.alumni.push(rec);
  call('saveAlumni',rec,function(res){if(res.ok){toast(EDIT_ROLL?'Updated':'Added','ok');closeM('m-alm');render();badge();}else toast('Failed: '+res.error,'bad');});
}

  DB.alumni=DB.alumni.filter(function(a){return a['Roll No']!=roll;});
  DB.donations=DB.donations.filter(function(d){return d.alumniRoll!=roll;});
  DB.attendance=DB.attendance.filter(function(a){return a.alumniRoll!=roll;});
  DB.jobHistory=DB.jobHistory.filter(function(h){return h.alumniRoll!=roll;});
  call('deleteAlumni',{roll:roll},function(res){if(res.ok){toast('Deleted','inf');render();badge();}});
}

function saveJobChange(){
  var a=DB.alumni.filter(function(x){return x['Roll No']==JC_ROLL;})[0];if(!a)return;
  var today=new Date().toISOString().split('T')[0];
  var hist={_id:uid(),alumniRoll:JC_ROLL,changeDate:gi('jc-date').value||today,
    designation:gi('jc-desig').value.trim(),organization:gi('jc-org').value.trim(),
    city:gi('jc-city').value.trim(),state:gi('jc-state').value.trim(),country:gi('jc-country').value.trim(),
    sector:gi('jc-sector').value,joiningDate:gi('jc-joining').value.trim(),notes:gi('jc-notes').value.trim()};
  var payload={roll:JC_ROLL,designation:hist.designation,organization:hist.organization,
    city:hist.city,state:hist.state,country:hist.country,sector:hist.sector,joiningDate:hist.joiningDate,history:hist};
  var idx=DB.alumni.findIndex(function(x){return x['Roll No']==JC_ROLL;});
  if(idx>=0){
    if(hist.designation)DB.alumni[idx]['Designation']=hist.designation;
    if(hist.organization)DB.alumni[idx]['Organization/Institute/University']=hist.organization;
    if(hist.city)DB.alumni[idx]['City']=hist.city;
    if(hist.state)DB.alumni[idx]['State/ Province']=hist.state;
    if(hist.country)DB.alumni[idx]['Country']=hist.country;
    if(hist.sector)DB.alumni[idx]['Sector']=hist.sector;
    if(hist.joiningDate)DB.alumni[idx]['Month and Year of Joining']=hist.joiningDate;
    DB.alumni[idx]['_lastVerified']=today;
  }
  DB.jobHistory.push(hist);
  call('saveJobChange',payload,function(res){if(res.ok){toast('Change logged','ok');closeM('m-jc');render();badge();}else toast('Failed: '+res.error,'bad');});
}


function doVerify(roll){
  var today=new Date().toISOString().split('T')[0];
  var idx=DB.alumni.findIndex(function(x){return x['Roll No']==roll;});

  call('markVerified',{roll:roll},function(res){if(res.ok){toast('Marked as verified','ok');render();badge();}});
}

var CSV_COLS=[
  'YOG','Roll No','Course','Name of the alumni','Designation',
  'Organization/Institute/University','Sector','City','State/ Province',
  'Country','Month and Year of Joining','Linkedin','IITGn Email',
  'Personal Email ID','Work Email ID',
  '_phone','_dob','_gender','_hometown','_marital','_friend','_awards'
];
var CSV_LABELS=[
  'YOG','Roll No','Course','Name','Designation','Organization','Sector',
  'City','State','Country','Joining Date','LinkedIn','IITGn Email',
  'Personal Email','Work Email','Phone','DOB','Gender','Hometown',
  'Marital Status','Emergency Contact','Awards'
];


function buildCSV(rows){
  var lines=[CSV_LABELS.map(function(l){return '"'+l+'"';}).join(',')];
  rows.forEach(function(a){
    lines.push(CSV_COLS.map(function(c){
      return '"'+String(a[c]||'').replace(/"/g,'""')+'"';
    }).join(','));
  });
  return lines.join('\n');
}
function triggerDownload(csv,filename){
  var el=document.createElement('a');
  el.href='data:text/csv;charset=utf-8,'+encodeURIComponent(csv);
  el.download=filename;el.click();
}


// Download all alumni (sidebar export)
function doExport(){
  triggerDownload(buildCSV(DB.alumni),'all_alumni_'+new Date().toISOString().slice(0,10)+'.csv');
  toast('Downloaded all '+DB.alumni.length+' alumni','ok');
}


// Download currently filtered/searched alumni
function downloadFilteredCSV(source){
  var rows;
  if(source==='search'){
    // From Advanced Search
    if(AI_MODE==='ai'&&AI_RESULT){
      var rs={};AI_RESULT.rolls.forEach(function(r){rs[r]=true;});
      rows=DB.alumni.filter(function(a){return rs[a['Roll No']];});
    } else {
      rows=applyManualFilters();
    }
  } else {
    // From Alumni Directory — apply current search + sort
    rows=DB.alumni.slice();
    if(SQ){
      var q=SQ.toLowerCase();
      rows=rows.filter(function(a){
        return[a['Name of the alumni'],a['Roll No'],a['Organization/Institute/University'],
          a['Designation'],a['City'],a['YOG']+'',a['Course']].join(' ').toLowerCase().indexOf(q)>=0;
      });
    }
    rows.sort(function(a,b){return((a[SORT.col]||'')>(b[SORT.col]||'')?1:-1)*SORT.dir;});
  }
  if(!rows.length){toast('No alumni to download','bad');return;}
  var label=source==='search'
    ?(AI_MODE==='ai'&&AI_QUERY?'ai_search_'+AI_QUERY.replace(/\s+/g,'_').slice(0,30):'filtered_alumni')
    :(SQ?'search_'+SQ.replace(/\s+/g,'_').slice(0,30):'alumni_directory');
  triggerDownload(buildCSV(rows),label+'_'+new Date().toISOString().slice(0,10)+'.csv');
  toast('Downloaded '+rows.length+' alumni ✓','ok');
}
